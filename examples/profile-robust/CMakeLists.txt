cmake_minimum_required(VERSION 3.20)
project(lyflow_profile_robust_consumer LANGUAGES CXX)

find_package(lyflow 2.0.1 EXACT REQUIRED COMPONENTS profile_robust_kernel)

add_executable(lyflow-profile-robust main.cpp)
target_compile_features(lyflow-profile-robust PRIVATE cxx_std_17)
target_link_libraries(lyflow-profile-robust PRIVATE lyflow::profile_robust_kernel)
if(MSVC)
  target_compile_options(lyflow-profile-robust PRIVATE /EHsc /utf-8 /W4)
endif()

set(_provider_dirs "${PCL_DIR}/../../bin")
get_property(_imported DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(_target IN LISTS _imported)
  foreach(_configuration "" _RELEASE _RELWITHDEBINFO)
    get_target_property(_location "${_target}" "IMPORTED_LOCATION${_configuration}")
    if(_location AND _location MATCHES "[.]dll$")
      get_filename_component(_directory "${_location}" DIRECTORY)
      list(APPEND _provider_dirs "${_directory}")
    endif()
  endforeach()
endforeach()
list(REMOVE_DUPLICATES _provider_dirs)
string(REPLACE ";" "\n" _provider_dirs "${_provider_dirs}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/provider-binary-dirs.txt" "${_provider_dirs}\n")

enable_testing()
add_test(NAME profile-robust COMMAND lyflow-profile-robust)
